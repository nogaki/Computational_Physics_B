\begin{frame}[t]{負の整数：2の補数}
  \begingroup
  \setlength{\parskip}{1mm}
  \setlength{\abovedisplayskip}{4pt}
  \setlength{\belowdisplayskip}{4pt}
  負の数も同じ足し算で扱うため、$-5$ は $5$ と足して $0$ になるビット列で表します。
  8~bitでは $256-5=251$ を足します。
  \[
    \renewcommand{\arraystretch}{1.35}
    \begin{array}{r@{\hspace{2.5mm}}r@{\hspace{2.5mm}}r}
      & & \text{\codefont\large 00000101} \\
      & \text{\codefont\large $+$} & \text{\codefont\large 11111011} \\ \cline{3-3}
      \textcolor{red!70!black}{\text{\codefont\large 1}} & & \text{\codefont\large \underline{00000000}}
    \end{array}
  \]
  \begin{center}
    9桁目への繰り上がり $1$ は捨て、\underline{下位8~bit}が残ります。
  \end{center}
  \vfill
  \begin{lessonbox}
    \code{11111011} は、符号なしでは $251$、2の補数では $-5$ を表します。
  \end{lessonbox}
  \endgroup
\end{frame}

\begin{frame}[t]{2の補数の値の読み方}
  \begingroup
  \setlength{\parskip}{0.2mm}
  \setlength{\abovedisplayskip}{2pt}
  \setlength{\belowdisplayskip}{2pt}
  前のページの \code{11111011} は、符号なしでは $251$、2の補数では $-5$ でした。
  $251-256=-5$ です。8~bitの2の補数では、先頭が $1$ なら負と読み、
  符号なしの値から $256$ を引きます。
  \begin{center}
  \begin{tikzpicture}[x=1mm,y=1mm]
    \foreach \i/\b/\w in {0/1/-128,1/1/64,2/1/32,3/1/16,4/1/8,5/0/4,6/1/2,7/1/1}{
      \fill[courseblue!8] (12*\i,0) rectangle (12*\i+11,6);
      \draw[courseblue!45,line width=.4pt] (12*\i,0) rectangle (12*\i+11,6);
      \node[font=\codefont\large] at (12*\i+5.5,3) {\b};
      \node[font=\scriptsize] at (12*\i+5.5,-1.1) {\w};
    }
  \end{tikzpicture}
  \end{center}
  \vspace{-1mm}
  {\small 最上位bitの重みは $128-256=-128$ です。}\par\noindent
  2の補数としての値は、先頭bitを $b_7$ として
  \[
    \mathrm{val}(b_7b_6\cdots b_0)=-b_7 2^7+\sum_{k=0}^{6} b_k2^k
  \]
  \vfill
  \begin{lessonbox}
    $M$ bitの2の補数の範囲は $-2^{M-1}$ から $2^{M-1}-1$ です。
  \end{lessonbox}
  \endgroup
\end{frame}

\begin{frame}[t,fragile]{0.1の表現を確認する}
  \setlength{\parskip}{0pt}
  0.1 は2進数で循環するため、有限桁に丸めて保存します。
  \[
    0.1=(0.0001100110011\ldots)_2
  \]
  \code{float}に保存されている値は、この丸めた値です。
\begin{lstlisting}
a = 0.1
print(f"{a:.55f}")
\end{lstlisting}
  \vspace{2mm}
  表示例：
  \[
    0.10000000000000000555\ldots
  \]
  \begin{lessonbox}
    表示された余分な桁は、計算機が近似値を保持していることを示します。
  \end{lessonbox}
\end{frame}

\begin{frame}[t,fragile]{繰り返し加算と丸め誤差}
  同じ小数を繰り返し足しても、数学上の値と計算結果は完全には一致しないことがあります。
\begin{lstlisting}
total = 0.0
for _ in range(10_000_000):
    total += 0.01
print(total)
\end{lstlisting}
  {\small 出力：\code{99999.99998630969}（$100000$ との差 約$1.4\times10^{-5}$）\par}
  数値計算では、丸め誤差が反復回数とともに積み重なる場合があります。
  \begin{itemize}
    \item 精度が必要な総和には、補償和などの方法もあります。
    \item まずは結果の桁数や収束を確かめます。
  \end{itemize}
\end{frame}

\begin{frame}[t,fragile]{テイラー展開から差分公式へ}
  \vspace*{\fill}
  $f$が十分滑らかなとき、
  \[
    f(x+h)=f(x)+hf'(x)+\frac{h^2}{2}f''(x)+O(h^3).
  \]
  $h$を小さくして差をとると、導関数を近似できます。
\begin{lstlisting}
(f(x + h) - f(x)) / h
\end{lstlisting}
  \vspace{1mm}
  ただし、$h$を小さくしすぎると、丸め誤差が増えます。
  \vfill
\end{frame}

\begin{frame}[t]{前進差分・後退差分・中心差分}
  \setlength{\parskip}{1mm}
  \[
    D_+f(x)=\frac{f(x+h)-f(x)}{h}=f'(x)+O(h)
  \]
  \[
    D_-f(x)=\frac{f(x)-f(x-h)}{h}=f'(x)+O(h)
  \]
  \[
    D_0f(x)=\frac{f(x+h)-f(x-h)}{2h}=f'(x)+O(h^2)
  \]
  \begin{itemize}
    \item 前進・後退差分：評価点の片側だけを使う。
    \item 中心差分：両側を使い、滑らかな関数では打ち切り誤差が小さい。
  \end{itemize}
\end{frame}

\begin{frame}[t]{二階微分の中心差分}
  前進差分・中心差分と同じように、左右対称な3点から二階微分を近似できます。
  \[
    f''(x)=\frac{f(x+h)-2f(x)+f(x-h)}{h^2}+O(h^2)
  \]
  \begin{itemize}
    \item 分子は、一次の変化が相殺されるように組み合わせています。
    \item 十分滑らかな関数では、打ち切り誤差は$O(h^2)$です。
  \end{itemize}
\end{frame}

\begin{frame}[t]{刻み幅 $h$ と2種類の誤差}
  \vspace*{\fill}
  \begin{columns}[t,onlytextwidth]
    \begin{column}{.48\textwidth}
      \textbf{打ち切り誤差}
      \begin{itemize}
        \item 差分公式を有限個の項で近似する誤差。
        \item $h$を小さくすると減る。
      \end{itemize}
    \end{column}
    \begin{column}{.48\textwidth}
      \textbf{丸め誤差}
      \begin{itemize}
        \item 浮動小数点演算による誤差。
        \item 差を$h$で割るため、小さい$h$で増えやすい。
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{2mm}
  \begin{lessonbox}
    誤差全体は$h$に対して谷を作ります。小さすぎず、大きすぎない刻み幅を探します。
  \end{lessonbox}
  \vfill
\end{frame}

\begin{frame}[t]{$\sin x$の数値微分}
  \vspace{-4mm}
  $x=0.3\pi$での真の微分係数は $\cos(0.3\pi)$ です。
  {\centering
    \includegraphics[width=.9\linewidth,height=44mm,keepaspectratio]{differentiation_error.pdf}\par}
  \small 前進差分の谷は$h\approx10^{-8}$、中心差分は$h\approx10^{-5}$付近にあります。

  \small Notebookで前進差分の誤差の谷を確認し、\\
  実習で中心差分と比べます。
\end{frame}

\begin{frame}[t]{刻み幅の目安}
  倍精度浮動小数点数の機械精度は $\epsilon\approx 2.2\times10^{-16}$ です。
  \begin{itemize}
    \setlength{\itemsep}{3mm}
    \item 前進差分では、打ち切り誤差$O(h)$と丸め誤差$O(\epsilon/h)$が競合します。
    \item 中心差分では、打ち切り誤差$O(h^2)$と丸め誤差$O(\epsilon/h)$が競合します。
  \end{itemize}
  \[
    h_{\rm forward}\sim \epsilon^{1/2}\approx 10^{-8},\qquad
    h_{\rm central}\sim \epsilon^{1/3}\approx 10^{-5}
  \]
  \small これは関数や評価点に依存する目安で、普遍的な最適値ではありません。
\end{frame}

\begin{frame}[t,fragile]{実習タイム：多項式の微分}
  \setlength{\parskip}{0pt}
  \vspace*{\fill}
  \begin{enumerate}
    \item $f(x)=x^3-2x+1$の解析解は$f'(x)=3x^2-2$です。
  \end{enumerate}
\begin{lstlisting}
def f(x):
    return x**3 - 2*x + 1

def df_exact(x):
    return 3*x*x - 2
\end{lstlisting}
  \vspace{2mm}
  \begin{itemize}
    \item $x=0.7$で差分近似を計算し、解析解と比べます。
    \item $h$を変え、前進差分・中心差分の誤差を記録します。
  \end{itemize}
  \vfill
\end{frame}

\begin{frame}[t]{実習の準備：関数が速く変化する場合}
  \setlength{\parskip}{0.5mm}
  $f(x)=\sin(\omega x)$では、導関数の大きさは$\omega$に比例し、高階導関数はさらに大きくなります。
  \[
    f'(x)=\omega\cos(\omega x),\qquad f'''(x)=-\omega^3\cos(\omega x)
  \]
  \begin{itemize}
    \item 打ち切り誤差の係数が変わるため、最適な$h$も変わります。
  \end{itemize}
  \vspace{1mm}
  \textbf{同位相で比べる}：$\sin x$は$x=0.3\pi$、$\sin(10^4x)$は$x=0.3\pi/10^4$。
  相対誤差は$|\text{近似値}-\text{解析解}|/|\text{解析解}|$。
  \vspace{0.5mm}

  $\omega$による最適な$h$の違いは、次のページで確かめます。
\end{frame}

\begin{frame}[t]{実習タイム（続き）：振動関数と差分公式}
  \begin{enumerate}
    \setlength{\itemsep}{4mm}
    \setcounter{enumi}{1}
    \item $\sin(10^4x)$を数値微分し、$\sin x$の場合と$h$の最適域を比べます。絶対誤差に加えて、微分係数の大きさに対する相対誤差も見ます。
    \item 前進差分・後退差分・中心差分を同じグラフに描き、$h$を極端に小さくしたときの桁落ちも確認します。
  \end{enumerate}
  \vfill
  \begin{lessonbox}
    解析解を持つ関数で、近似値と誤差の両方を確かめましょう。
  \end{lessonbox}
\end{frame}

\begin{frame}[t,fragile]{実習タイム（続き）：丸め誤差の積み重ね}
  \setlength{\parskip}{0pt}
  \vspace*{\fill}
  \begin{enumerate}
    \setcounter{enumi}{3}
    \item 0.01の反復加算で、加算回数$n$を$10^4$、$10^5$、$10^6$と変え、$n\times0.01$との差を比べます。
  \end{enumerate}
\begin{lstlisting}
n = 10**4
total = 0.0
for _ in range(n):
    total += 0.01
print(n * 0.01 - total)
\end{lstlisting}
  \vspace{1mm}
  \code{n}を$10^5$、$10^6$に変えて、差の増え方を比べます。
  \vfill
\end{frame}

\begin{frame}[t,fragile]{実習タイム（続き）：二階微分の中心差分}
